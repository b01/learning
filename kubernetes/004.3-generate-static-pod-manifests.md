# Generate Static Pod Manifests

1. Set the following environment variables on `control-plane-01`.

   ```shell
   getIps () {
      iface=${1}
      export IPV4=$(ip -4 addr show scope global dev ${iface} | grep "inet\b" | awk '{print $2}' | cut -d/ -f1)
      export IPV6=$(ip -6 addr show scope global dev ${iface} | grep "inet6\b" | awk '{print $2}' | cut -d/ -f1)
   }
   getIps "enp0s8"
   export ADVERTISE_ADDRESS=${IPV6}
   export ADVERTISE_ADDRESS_URL="[${IPV6}]"
   export POD_CIDRS=fd12:3456:789a:1000::/56,10.244.0.0/16
   export SERVICE_CIDRS=fd12:3456:789a:2000::/112,10.96.0.0/16
   export ETCD_IMG_VER=3.6.12-0
   export KUBE_IMG_VER=v1.35.6
   export HOSTNAME=$(hostname)
   sudo mkdir -p /etc/kubernetes/manifests/
   mkdir -p ~/manifests
   cd ~/manifests
   cp /vagrant/*.yaml.tmpl .
   ```
2. Copy the YAML files to `control-plane-01`.

   ```shell
   envsubst < etcd.yaml.tmpl | sudo tee /etc/kubernetes/manifests/etcd.yaml
   envsubst < /vagrant/kube-controller-manager.yaml.tmpl | sudo tee /etc/kubernetes/manifests/kube-controller-manager.yaml
   envsubst < /vagrant/kube-apiserver.yaml.tmpl | sudo tee /etc/kubernetes/manifests/kube-apiserver.yaml
   envsubst < /vagrant/kube-scheduler.yaml.tmpl | sudo tee /etc/kubernetes/manifests/kube-scheduler.yaml
   ```
3. Test out certain components are working:

   ```shell
   sudo curl -6 -v https://[2000:192:168:56::11]:2379/health \
       --cacert /etc/kubernetes/pki/etcd/ca.crt \
       --cert /etc/kubernetes/pki/apiserver-etcd-client.crt \
       --key /etc/kubernetes/pki/apiserver-etcd-client.key
   # Test with the etcd peer key-pair.
   sudo curl -6 -v https://[2000:192:168:56::11]:2379/health \
       --cacert /etc/kubernetes/pki/etcd/ca.crt \
       --cert /etc/kubernetes/pki/etcd/peer.crt \
       --key /etc/kubernetes/pki/etcd/peer.key
   # Test kube-apiserver endpoint
   sudo curl -6 -v https://[2000:192:168:56::11]:6443/readyz \
       --cacert /etc/kubernetes/pki/ca.crt \
       --cert /etc/kubernetes/pki/apiserver.crt \
       --key /etc/kubernetes/pki/apiserver.key
   # Test 2381 in-secure endpoints no key-pair required.
   sudo crictl ps | grep  etcd
   sudo crictl inspect 88c0377cfb077 | grep -i pid
   sudo nsenter -t 20500 -n curl -6 -v http://[::1]:2381/readyz
   #
   sudo crictl ps | grep kube-apiserver
   sudo crictl logs 3715cc277d056 | grep -i etcd

   # find the PID of kube-apiserver
   sudo crictl inspect 3715cc277d056 | grep -i pid
   # enter the container’s network namespace
   sudo nsenter -t 15160 -n bash
   sudo nsenter -t 15160 -n curl -6 -v https://[2000:192:168:56::11]:2379/health

   curl -6 -v https://[2000:192:168:56::11]:2379/health \
      --cacert /etc/kubernetes/pki/etcd/ca.crt \
      --cert /etc/kubernetes/pki/apiserver-etcd-client.crt \
      --key /etc/kubernetes/pki/apiserver-etcd-client.key

   sudo crictl exec -it 3715cc277d056 -- sh -c \
      "curl -6 -v https://[2000:192:168:56::11]:2379/health \
      --cacert /etc/kubernetes/pki/etcd/ca.crt \
      --cert /etc/kubernetes/pki/apiserver-etcd-client.crt \
      --key /etc/kubernetes/pki/apiserver-etcd-client.key"
   ```
